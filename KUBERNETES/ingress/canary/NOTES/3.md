You can use **here documents (`cat <<EOF`)** to create all the manifest files in one command sequence.

```bash
mkdir -p canary-demo && cd canary-demo

cat > 01-v1-deployment.yaml <<'EOF'
apiVersion: apps/v1
kind: Deployment
metadata:
  name: blogging-v1
  labels:
    app: blogging
    version: v1
spec:
  replicas: 3
  selector:
    matchLabels:
      app: blogging
      version: v1
  template:
    metadata:
      labels:
        app: blogging
        version: v1
    spec:
      containers:
      - name: nginx
        image: nginx:1.27
        ports:
        - containerPort: 80
        command: ["/bin/sh","-c"]
        args:
        - |
          echo "<h1>VERSION 1</h1>" > /usr/share/nginx/html/index.html
          nginx -g 'daemon off;'
EOF

cat > 02-v1-service.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: blogging-v1-svc
spec:
  selector:
    app: blogging
    version: v1
  ports:
  - port: 80
    targetPort: 80
EOF

cat > 03-v1-ingress.yaml <<'EOF'
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: blogging-ingress
spec:
  ingressClassName: nginx
  rules:
  - host: blog.example.com
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: blogging-v1-svc
            port:
              number: 80
EOF

cat > 04-v2-deployment.yaml <<'EOF'
apiVersion: apps/v1
kind: Deployment
metadata:
  name: blogging-v2
  labels:
    app: blogging
    version: v2
spec:
  replicas: 3
  selector:
    matchLabels:
      app: blogging
      version: v2
  template:
    metadata:
      labels:
        app: blogging
        version: v2
    spec:
      containers:
      - name: nginx
        image: nginx:1.27
        ports:
        - containerPort: 80
        command: ["/bin/sh","-c"]
        args:
        - |
          echo "<h1>VERSION 2</h1>" > /usr/share/nginx/html/index.html
          nginx -g 'daemon off;'
EOF

cat > 05-v2-service.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: blogging-v2-svc
spec:
  selector:
    app: blogging
    version: v2
  ports:
  - port: 80
    targetPort: 80
EOF

cat > 06-v2-canary-ingress.yaml <<'EOF'
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: blogging-canary
  annotations:
    nginx.ingress.kubernetes.io/canary: "true"
    nginx.ingress.kubernetes.io/canary-weight: "30"
spec:
  ingressClassName: nginx
  rules:
  - host: blog.example.com
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: blogging-v2-svc
            port:
              number: 80
EOF
```

### Verify

```bash
ls -1
```

Output:

```text
01-v1-deployment.yaml
02-v1-service.yaml
03-v1-ingress.yaml
04-v2-deployment.yaml
05-v2-service.yaml
06-v2-canary-ingress.yaml
```

### Deploy Everything

Deploy the stable version first:

```bash
kubectl apply -f 01-v1-deployment.yaml -f 02-v1-service.yaml -f 03-v1-ingress.yaml
```

Then deploy the canary version:

```bash
kubectl apply -f 04-v2-deployment.yaml -f 05-v2-service.yaml -f 06-v2-canary-ingress.yaml
```

This gives you a complete working canary deployment using the NGINX Ingress Controller.
