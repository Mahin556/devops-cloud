```yaml
apiVersion: v1
kind: PersistentVolume
metadata:
  name: jhooq-demo-pv

spec:
  capacity:
    storage: 1Gi                 # Total storage size

  volumeMode: Filesystem         # Normal filesystem volume

  accessModes:
    - ReadWriteOnce              # One node can mount read-write

  persistentVolumeReclaimPolicy: Retain  # Data kept after PVC deletion

  storageClassName: local-storage        # Must match PVC

  local:
    path: /home/vagrant/storage  # Directory on node

  nodeAffinity:                  # Tells K8s which node this disk lives on
    required:
      nodeSelectorTerms:
      - matchExpressions:
        - key: kubernetes.io/hostname
          operator: In
          values:
          - node01
```
```yaml
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: jhooq-pvc

spec:
  volumeName: jhooq-demo-pv       # Bind to specific PV

  storageClassName: local-storage

  volumeMode: Filesystem

  accessModes:
    - ReadWriteOnce

  resources:
    requests:
      storage: 1Gi
```
```yaml
apiVersion: v1
kind: Pod
metadata:
  name: jhooq-pod-with-pvc
  labels:
    name: jhooq-pod-with-pvc

spec:
  nodeName: node01
  containers:
  - name: jhooq-pod-with-pvc
    image: rahulwagh17/kubernetes-jhooq-k8s-springboot

    ports:
      - containerPort: 8080
        name: www

    volumeMounts:
      - name: www-persistent-storage
        mountPath: /home/vagrant/storage   # Inside container

  volumes:
    - name: www-persistent-storage
      persistentVolumeClaim:
        claimName: jhooq-pvc
```
