```bash
kubectl apply -f https://raw.githubusercontent.com/kubernetes/ingress-nginx/controller-v1.14.3/deploy/static/provider/cloud/deploy.yaml

kubectl get pods -n ingress-nginx

kubectl create namespace myapp-ns

kubectl apply -f -<<EOF
apiVersion: apps/v1
kind: Deployment
metadata:
  name: myapp-deploy
  namespace: myapp-ns
spec:
  replicas: 2
  selector:
    matchLabels:
      app: myapp
  template:
    metadata:
      labels:
        app: myapp
    spec:
      containers:
      - name: myapp-container
        image: nginx:latest
        ports:
        - containerPort: 80
EOF

kubectl get pods -n myapp-ns

kubectl apply -f -<<EOF
apiVersion: v1
kind: Service
metadata:
  name: myapp-service
  namespace: myapp-ns
spec:
  selector:
    app: myapp
  ports:
    - port: 80
      targetPort: 80
  type: ClusterIP
EOF

kubectl get svc -n myapp-ns
kubectl get ep -n myapp-ns

openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
-keyout tls.key \
-out tls.crt \
-subj "/CN=myapp.local/O=myapp"

kubectl create secret tls myapp-tls \
--key tls.key \
--cert tls.crt \
-n myapp-ns

kubectl get secret -n myapp-ns
kubectl describe secret -n myapp-ns
kubectl get secret -n myapp-ns -oyaml

kubectl apply -f -<<EOF
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: myapp-ingress
  namespace: myapp-ns
  annotations:
    nginx.ingress.kubernetes.io/rewrite-target: /
spec:
  ingressClassName: nginx
  tls:
  - hosts:
    - myapp.local
    secretName: myapp-tls
  rules:
  - host: myapp.local
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: myapp-service
            port:
              number: 80
EOF

kubectl get ingress -n myapp-ns

kubectl patch svc ingress-nginx-controller \
-n ingress-nginx \
-p '{"spec": {"type": "NodePort"}}'

kubectl get all -n ingress-nginx

curl -k https://myapp.local:32595 --resolve myapp.local:32595:172.30.1.2

kubectl describe ingress myapp-ingress -n myapp-ns

kubectl get endpoints -n myapp-ns

kubectl logs -n ingress-nginx -l app.kubernetes.io/component=controller

kubectl exec -it -n ingress-nginx <controller-pod-name> -- cat /etc/nginx/nginx.conf
```
