          - node01

```bash
cat << EOF | kubectl apply -f -
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: local-sc
provisioner: kubernetes.io/no-provisioner
volumeBindingMode: WaitForFirstConsumer
EOF

mkdir -p /mnt/data/localvol

cat << EOF | kubectl apply -f -
apiVersion: v1
kind: PersistentVolume
metadata:
  name: my-local-pv
spec:
  capacity:
    storage: 1Gi
  accessModes:
    - ReadWriteOnce
  storageClassName: local-sc
  persistentVolumeReclaimPolicy: Retain
  local:
    path: /mnt/data/localvol
  nodeAffinity:
    required:
      nodeSelectorTerms:
      - matchExpressions:
        - key: kubernetes.io/hostname
          operator: In
          values:
          - node01
EOF

cat << EOF | kubectl apply -f -
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: my-claim
spec:
  storageClassName: local-sc
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 1Gi
EOF

kubectl get pv,pvc

cat << EOF | kubectl apply -f -
apiVersion: v1
kind: Pod
metadata:
  name: www
spec:
  containers:
  - name: app
    image: busybox
    command: ["sh", "-c", "echo Hello local persistent volume > /data/index.html; sleep 3600"]
    volumeMounts:
    - name: storage
      mountPath: /data
  volumes:
  - name: storage
    persistentVolumeClaim:
      claimName: my-claim
EOF

kubectl delete pod www
kubectl get pv,pvc #Even though Pod is gone, storage is still reserved(bounded) due to retain `persistentVolumeReclaimPolicy`.
```

#### Show that two Pods on the same node can write to the same RWO local volume.

```bash
# Write from Pod 1

kubectl exec -it centos-local-volume -- sh
echo "centos-local-volume has changed the content" > /data/index.html
exit

# Check Pod 2 logs

kubectl logs centos-local-volume2 | tail -n 3

# Output:
# centos-local-volume has changed the content

# Meaning: Pod 2 can see the file updated by Pod 1.

# Write from Pod 2

kubectl exec -it centos-local-volume2 -- sh
echo "centos-local-volume2 has changed the content" > /data/index.html
exit

# Check Pod 1 logs

kubectl logs centos-local-volume | tail -n 3

# Output:
# centos-local-volume2 has changed the content

# Meaning: Pod 1 now sees the update from Pod 2.


# • Both Pods mount the same filesystem
# • Writes are instantly visible between Pods
# • It is one shared disk, not replicated storage
# * AccessMode = ReadWriteOnce (RWO)
# • Volume can be mounted read-write
# • Multiple Pods can use it
# • But only on ONE node
# * Local PV keeps both Pods on the same node → shared writes allowed.
# * RWO = multi-Pod write allowed on the same node, app must manage safety.
```
