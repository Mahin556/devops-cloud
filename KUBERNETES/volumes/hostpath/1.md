```bash
ssh node01

DIRNAME="vol"
mkdir -p /mnt/data/$DIRNAME 
chmod 777 /mnt/data/$DIRNAME
echo "Hello persistent volume" > /mnt/data/vol/index.html

cat > vol.yaml << EOF
apiVersion: v1
kind: PersistentVolume
metadata:
  name: vol
spec:
  capacity:
    storage: 500Mi
  accessModes:
    - ReadWriteOnce
    - ReadWriteMany
  persistentVolumeReclaimPolicy: Retain
  hostPath:
    path: /mnt/data/vol
    type: Directory
  nodeAffinity:
    required:
      nodeSelectorTerms:
      - matchExpressions:
        - key: kubernetes.io/hostname
          operator: In
          values:
          - node01   #This binds storage to Node01
EOF

kubectl apply -f vol.yaml

cat > http-claim.yaml << EOF
kind: PersistentVolumeClaim
apiVersion: v1
metadata:
  name: http-claim
spec:
  storageClassName: ""
  accessModes:
    - ReadWriteOnce
    - ReadWriteMany
  resources:
    requests:
      storage: 500Mi
EOF

kubectl apply -f http-claim.yaml

controlplane:~$ kubectl get pv,pvc
NAME                   CAPACITY   ACCESS MODES   RECLAIM POLICY   STATUS   CLAIM                STORAGECLASS   VOLUMEATTRIBUTESCLASS   REASON   AGE
persistentvolume/vol   500Mi      RWO,RWX        Retain           Bound    default/http-claim                  <unset>                          3m22s

NAME                               STATUS   VOLUME   CAPACITY   ACCESS MODES   STORAGECLASS   VOLUMEATTRIBUTESCLASS   AGE
persistentvolumeclaim/http-claim   Bound    vol      500Mi      RWO,RWX                       <unset>                 2s


controlplane:~$ kubectl describe pvc http-claim
Name:          http-claim
Namespace:     default
StorageClass:

Labels:        <none>

               pv.kubernetes.io/bound-by-controller: yes
Finalizers:    [kubernetes.io/pvc-protection]

Access Modes:  RWO,RWX
VolumeMode:    Filesystem
Used By:       <none>
Events:        <none>

cat > http-pod.yaml << EOF
apiVersion: v1
kind: Pod
metadata:
  name: www
  labels:
    name: www
spec:
  nodeName: node01
  containers:
  - name: www
    image: nginx:alpine
    ports:
      - containerPort: 80
        name: www
    volumeMounts:
      - name: www-persistent-storage
        mountPath: /usr/share/nginx/html
  volumes:
    - name: www-persistent-storage
      persistentVolumeClaim:
        claimName: http-claim
EOF

kubectl create -f http-pod.yaml

curl <pod_ip>

cat <<EOF | kubectl create -f -
apiVersion: v1
kind: Pod
metadata:
  name: busybox
  labels:
    name: busybox
spec:
  nodeName: node01
  containers:
  - name: centos
    image: busybox
    command: ["/bin/sh"]
    args: ["-c", "while true; do cat /data/index.html; sleep 10; done"]
    volumeMounts:
      - name: www-persistent-storage
        mountPath: /data
  volumes:
    - name: www-persistent-storage
      persistentVolumeClaim:
        claimName: http-claim
EOF
```
