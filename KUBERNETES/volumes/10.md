```yaml
apiVersion: v1
kind: Pod
metadata:
  name: silly-demo
  labels:
    app: silly-demo

spec:
  containers:
  - name: silly-demo
    image: ghcr.io/vfarcic/silly-demo:1.4.235-alpine

    ports:
      - containerPort: 8080

    # Load ALL ConfigMap keys as environment variables
    envFrom:
      - configMapRef:
          name: silly-demo

    # Load single key as environment variable
    env:
      - name: MESSAGE
        valueFrom:
          configMapKeyRef:
            name: silly-demo
            key: message

    # Mount ConfigMap as files
    volumeMounts:
      - name: cache
        mountPath: /cache

  volumes:
    - name: cache
      configMap:
        name: silly-demo   # videos.yaml appears inside /cache
```
