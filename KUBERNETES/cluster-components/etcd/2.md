## Phase 0 — Prepare All ETCD Nodes

> Run on **etcd-1, etcd-2, etcd-3**

```bash
# Disable swap
swapoff -a
sed -i '/ swap / s/^\(.*\)$/#\1/g' /etc/fstab

# Kernel modules
cat << EOF > /etc/modules-load.d/k8s.conf
overlay
br_netfilter
EOF

modprobe overlay
modprobe br_netfilter

# Sysctl
cat << EOF > /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-iptables  = 1
net.ipv4.ip_forward                 = 1
net.bridge.bridge-nf-call-ip6tables = 1
EOF

sysctl --system
```

## Phase 1 — Install Software on All ETCD Nodes

> Run on **etcd-1, etcd-2, etcd-3**

### 1.1 Install containerd

```bash
apt-get update && \
apt-get install -y ca-certificates curl gnupg && \
install -m 0755 -d /etc/apt/keyrings && \
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | gpg --dearmor -o /etc/apt/keyrings/docker.gpg && \
chmod a+r /etc/apt/keyrings/docker.gpg && \
echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/ubuntu $(. /etc/os-release && echo "$VERSION_CODENAME") stable" | tee /etc/apt/sources.list.d/docker.list && \
apt-get update && \
apt-get install -y containerd.io && \
containerd config default > /etc/containerd/config.toml && \
sed -i 's/SystemdCgroup = false/SystemdCgroup = true/' /etc/containerd/config.toml && \
systemctl enable --now containerd
```

### 1.2 Install kubelet, kubeadm, kubectl

```bash
curl -fsSL https://pkgs.k8s.io/core:/stable:/v1.32/deb/Release.key | gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg && \
echo 'deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v1.32/deb/ /' | tee /etc/apt/sources.list.d/kubernetes.list && \
apt-get update && \
apt-get install -y kubelet kubeadm kubectl && \
apt-mark hold kubelet kubeadm kubectl
```

### 1.3 kubelet Service Override for ETCD Static Pods

```bash
mkdir -p /usr/lib/systemd/system/kubelet.service.d/

cat << EOF > /usr/lib/systemd/system/kubelet.service.d/20-etcd-service-manager.conf
[Service]
ExecStart=
ExecStart=/usr/bin/kubelet \
  --address=127.0.0.1 \
  --pod-manifest-path=/etc/kubernetes/manifests \
  --cgroup-driver=systemd \
  --runtime-request-timeout=15m \
  --container-runtime-endpoint=unix:///run/containerd/containerd.sock
Restart=always
EOF

systemctl daemon-reload && \
systemctl enable kubelet && \
systemctl restart kubelet
```

### 1.4 Generate kubeadm Config Files

> Run on **etcd-1 only**

```bash
cat << 'EOF' > /root/kubeadm_setup.sh
#!/bin/bash
export HOST0=192.168.29.98
export HOST1=192.168.29.105
export HOST2=192.168.29.88
export NAME0="etcd-1"
export NAME1="etcd-2"
export NAME2="etcd-3"

mkdir -p /tmp/${HOST0}/ /tmp/${HOST1}/ /tmp/${HOST2}/

HOSTS=(${HOST0} ${HOST1} ${HOST2})
NAMES=(${NAME0} ${NAME1} ${NAME2})

for i in "${!HOSTS[@]}"; do
  HOST=${HOSTS[$i]}
  NAME=${NAMES[$i]}
  cat << YAML > /tmp/${HOST}/kubeadmcfg.yaml
---
apiVersion: "kubeadm.k8s.io/v1beta3"
kind: InitConfiguration
nodeRegistration:
  name: ${NAME}
  criSocket: unix:///run/containerd/containerd.sock
localAPIEndpoint:
  advertiseAddress: ${HOST}
---
apiVersion: "kubeadm.k8s.io/v1beta3"
kind: ClusterConfiguration
etcd:
  local:
    serverCertSANs:
    - "${HOST}"
    peerCertSANs:
    - "${HOST}"
    extraArgs:
      initial-cluster: ${NAMES[0]}=https://${HOSTS[0]}:2380,${NAMES[1]}=https://${HOSTS[1]}:2380,${NAMES[2]}=https://${HOSTS[2]}:2380
      initial-cluster-state: new
      name: ${NAME}
      listen-peer-urls: https://${HOST}:2380
      listen-client-urls: https://${HOST}:2379
      advertise-client-urls: https://${HOST}:2379
      initial-advertise-peer-urls: https://${HOST}:2380
YAML
done
EOF

chmod +x /root/kubeadm_setup.sh && ./kubeadm_setup.sh
```

## Phase 2 — Generate TLS Certificates

> Run on **etcd-1 only**

### 2.1 Generate the ETCD CA

```bash
kubeadm init phase certs etcd-ca && \
tree /etc/kubernetes/pki/etcd/
```

### 2.2 Generate Per-Node Certificates

```bash
export HOST0=192.168.29.98
export HOST1=192.168.29.105
export HOST2=192.168.29.88

# --- etcd-3 (HOST2) ---
kubeadm init phase certs etcd-server             --config=/tmp/${HOST2}/kubeadmcfg.yaml && \
kubeadm init phase certs etcd-peer               --config=/tmp/${HOST2}/kubeadmcfg.yaml && \
kubeadm init phase certs etcd-healthcheck-client --config=/tmp/${HOST2}/kubeadmcfg.yaml && \
kubeadm init phase certs apiserver-etcd-client   --config=/tmp/${HOST2}/kubeadmcfg.yaml && \
cp -R /etc/kubernetes/pki /tmp/${HOST2}/ && \
find /etc/kubernetes/pki -not -name ca.crt -not -name ca.key -type f -delete

# --- etcd-2 (HOST1) ---
kubeadm init phase certs etcd-server             --config=/tmp/${HOST1}/kubeadmcfg.yaml && \
kubeadm init phase certs etcd-peer               --config=/tmp/${HOST1}/kubeadmcfg.yaml && \
kubeadm init phase certs etcd-healthcheck-client --config=/tmp/${HOST1}/kubeadmcfg.yaml && \
kubeadm init phase certs apiserver-etcd-client   --config=/tmp/${HOST1}/kubeadmcfg.yaml && \
cp -R /etc/kubernetes/pki /tmp/${HOST1}/ && \
find /etc/kubernetes/pki -not -name ca.crt -not -name ca.key -type f -delete

# --- etcd-1 (HOST0, local) ---
kubeadm init phase certs etcd-server             --config=/tmp/${HOST0}/kubeadmcfg.yaml && \
kubeadm init phase certs etcd-peer               --config=/tmp/${HOST0}/kubeadmcfg.yaml && \
kubeadm init phase certs etcd-healthcheck-client --config=/tmp/${HOST0}/kubeadmcfg.yaml && \
kubeadm init phase certs apiserver-etcd-client   --config=/tmp/${HOST0}/kubeadmcfg.yaml

# Remove ca.key from remote bundles
find /tmp/${HOST2} -name ca.key -type f -delete && \
find /tmp/${HOST1} -name ca.key -type f -delete
```

### 2.3 Distribute Certificates to etcd-2 and etcd-3

```bash
export HOST1=192.168.29.105
export HOST2=192.168.29.88
# Copy to etcd-2
scp -r /tmp/${HOST1}/* root@${HOST1}: && \
ssh root@${HOST1} 'chown -R root:root pki && mv pki /etc/kubernetes/'

# Copy to etcd-3
scp -r /tmp/${HOST2}/* root@${HOST2}: && \
ssh root@${HOST2} 'chown -R root:root pki && mv pki /etc/kubernetes/'
```

## Phase 3 — Bootstrap the ETCD Cluster

### 3.1 Create Static Pod Manifests

> Run on **each etcd node** using its own config

```bash
# On etcd-1
kubeadm init phase etcd local --config=/tmp/192.168.29.98/kubeadmcfg.yaml

# On etcd-2
kubeadm init phase etcd local --config=/tmp/192.168.29.105/kubeadmcfg.yaml

# On etcd-3
kubeadm init phase etcd local --config=/tmp/192.168.29.88/kubeadmcfg.yaml
```

### 3.2 Install etcdctl

```bash
ETCD_RELEASE=$(curl -s https://api.github.com/repos/etcd-io/etcd/releases/latest | grep tag_name | cut -d '"' -f 4) && \
curl -LO https://github.com/etcd-io/etcd/releases/download/${ETCD_RELEASE}/etcd-${ETCD_RELEASE}-linux-amd64.tar.gz && \
tar zxvf etcd-${ETCD_RELEASE}-linux-amd64.tar.gz && \
cp etcd-${ETCD_RELEASE}-linux-amd64/etcdctl /usr/local/bin && \
etcdctl version
```

### 3.3 Verify Cluster Health

```bash

ETCDCTL_API=3 etcdctl \
  --cert    /etc/kubernetes/pki/etcd/peer.crt \
  --key     /etc/kubernetes/pki/etcd/peer.key \
  --cacert  /etc/kubernetes/pki/etcd/ca.crt \
  --endpoints https://192.168.29.98:2379,https://192.168.29.105:2379,https://192.168.29.88:2379 \
  endpoint health

ETCDCTL_API=3 etcdctl \
  --cert    /etc/kubernetes/pki/etcd/peer.crt \
  --key     /etc/kubernetes/pki/etcd/peer.key \
  --cacert  /etc/kubernetes/pki/etcd/ca.crt \
  --endpoints https://192.168.29.98:2379,https://192.168.29.105:2379,https://192.168.29.88:2379 \
```

## Phase 4 — Integrate with Kubernetes Control Plane

### 4.1 Copy ETCD Certificates to master-1 and master-2

> Run on **etcd-1**

```bash
export MASTER1=192.168.29.69
export MASTER2=192.168.29.16

# Create dirs on master nodes
ssh root@$MASTER1 "mkdir -p /etc/kubernetes/pki/etcd/" && \
ssh root@$MASTER2 "mkdir -p /etc/kubernetes/pki/etcd/" && \

# Copy to master-1
scp /etc/kubernetes/pki/etcd/ca.crt              root@$MASTER1:/etc/kubernetes/pki/etcd/ && \
scp /etc/kubernetes/pki/apiserver-etcd-client.crt root@$MASTER1:/etc/kubernetes/pki/ && \
scp /etc/kubernetes/pki/apiserver-etcd-client.key root@$MASTER1:/etc/kubernetes/pki/ && \

# Copy to master-2
scp /etc/kubernetes/pki/etcd/ca.crt              root@$MASTER2:/etc/kubernetes/pki/etcd/ && \
scp /etc/kubernetes/pki/apiserver-etcd-client.crt root@$MASTER2:/etc/kubernetes/pki/ && \
scp /etc/kubernetes/pki/apiserver-etcd-client.key root@$MASTER2:/etc/kubernetes/pki/
```

### 4.2 Install containerd + kubeadm on Master Nodes

> Run on **master-1 and master-2** — same as Phase 1.1 and 1.2 above, but **skip** the kubelet override (1.3)

### 4.3 Create kubeadm Config on master-1

```bash
export HOST0=192.168.29.98
export HOST1=192.168.29.105
export HOST2=192.168.29.88
export LB=192.168.29.203

cat <<EOF > /root/kubeadm-config.yaml
apiVersion: kubeadm.k8s.io/v1beta3
kind: ClusterConfiguration
kubernetesVersion: stable
apiServer:
  certSANs:
  - "${LB}"
controlPlaneEndpoint: "${LB}:6443"
etcd:
  external:
    endpoints:
    - https://${HOST0}:2379
    - https://${HOST1}:2379
    - https://${HOST2}:2379
    caFile:   /etc/kubernetes/pki/etcd/ca.crt
    certFile: /etc/kubernetes/pki/apiserver-etcd-client.crt
    keyFile:  /etc/kubernetes/pki/apiserver-etcd-client.key
networking:
  podSubnet: "10.244.0.0/16"
---
apiVersion: kubeadm.k8s.io/v1beta3
kind: InitConfiguration
nodeRegistration:
  criSocket: unix:///run/containerd/containerd.sock
EOF
```

### 4.4 Initialize the Cluster

```bash
# On master-1
kubeadm init --config /root/kubeadm-config.yaml --upload-certs
```

### 4.5 Configure kubectl

```bash
mkdir -p $HOME/.kube && \
cp -i /etc/kubernetes/admin.conf $HOME/.kube/config && \
chown $(id -u):$(id -g) $HOME/.kube/config && \
kubectl get nodes && \
kubectl get pods -n kube-system
```

### 4.6 Join master-2

> Use the `--control-plane` join command printed by `kubeadm init` output:

```bash
kubeadm join 192.168.29.203:6443 \
  --token <token> \
  --discovery-token-ca-cert-hash sha256:<hash> \
  --control-plane \
  --certificate-key <cert-key>
```
