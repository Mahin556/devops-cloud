  namespace: myapp

  replicas: 2
  selector:
    matchLabels:
      app: nginx
  template:
    metadata:
      labels:
        app: nginx
    spec:
      containers:
      - name: nginx
        image: nginx
        ports:
        - containerPort: 80
EOF

kubectl get deploy -n myapp

kubectl apply -f -<<EOF
apiVersion: v1
kind: Service
metadata:
  name: nginx-service
  namespace: myapp
spec:
  selector:
    app: nginx
  ports:
    - port: 80
      targetPort: 80
  type: ClusterIP
EOF

kubectl get svc -n myapp


kubectl apply -f -<<EOF
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: nginx-ingress
  namespace: myapp
spec:
  ingressClassName: nginx
  rules:
  - host: myapp.local.com
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: nginx-service
            port:
              number: 80
EOF

curl http://myapp.local.com:80 --resolve myapp.local.com:80:152.42.159.172

kubectl delete ing/nginx-ingress svc/nginx-service deploy/nginx-deployment ns/myapp -n myapp

```bash
#TLS integration
openssl req -x509 -nodes -days 365 \
-newkey rsa:2048 \
-keyout tls.key \
-out tls.crt \
-subj "/CN=todo.com"

kubectl create secret tls todo-tls \
--key tls.key \
--cert tls.crt

kubectl apply -f -<<EOF
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: tls-ingress
  namespace: todo
spec:
  tls:
  - hosts:
    - todo.local
    secretName: todo-tls
  rules:
  - host: todo.local
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: todo-ui-service
            port:
              number: 8080
EOF

### Create Default Backend Deployment
```bash
kubectl apply -f -<<EOF
apiVersion: apps/v1
kind: Deployment
metadata:
  name: default-backend
  namespace: todo
spec:
  replicas: 1
  selector:
    matchLabels:
      app: default-backend
  template:
    metadata:
      labels:
        app: default-backend
    spec:
      containers:
      - name: default-backend
        image: hashicorp/http-echo
        args:
        - "-text=404 Page Not Found"
        ports:
        - containerPort: 5678
EOF

kubectl apply -f -<<EOF
apiVersion: v1
kind: Service
metadata:
  name: default-backend-service
  namespace: todo
spec:
  selector:
    app: default-backend
  ports:
  - port: 80
    targetPort: 5678
  type: ClusterIP
EOF

kubectl apply -f -<<EOF
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: my-ingress
  namespace: todo
spec:
  defaultBackend:
    service:
      name: default-backend-service
      port:
        number: 80
  rules:
  - host: myapp.local
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: api-service
            port:
              number: 80
EOF
