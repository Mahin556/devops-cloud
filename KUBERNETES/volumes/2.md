```yaml
apiVersion: v1
kind: PersistentVolume
metadata:
  name: gce-pv
spec:
  capacity:
    storage: 100Gi
  accessModes:
    - ReadWriteOnce
  persistentVolumeReclaimPolicy: Delete
  storageClassName: gce-sc
  gcePersistentDisk:
    pdName: my-data-disk
    fsType: ext4
---
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: gce-pvc
spec:
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 100Gi
  storageClassName: gce-sc
---
apiVersion: v1
kind: Pod
metadata:
  name: gce-pod
spec:
  containers:
  - name: app
    image: nginx
    volumeMounts:
    - mountPath: /data
      name: gce-vol
  volumes:
  - name: gce-vol
    persistentVolumeClaim:
      claimName: gce-pvc
```
