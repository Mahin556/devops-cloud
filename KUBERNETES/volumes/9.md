```yaml
apiVersion: v1
kind: PersistentVolume
metadata:
  name: fci-solution-logs
  labels:                    # These labels satisfy the PVC selector
    release: stable
    environment: dev
spec:
  capacity:
    storage: 1Gi             # Must be >= PVC request
  accessModes:
    - ReadWriteOnce          # Must support PVC mode
  persistentVolumeReclaimPolicy: Retain  # Data kept after PVC deletion
  storageClassName: slow     # Must match PVC SC
  nfs:
    path: /fci-solution       # Real storage path on NFS server
    server: HOSTNAME          # NFS server address

---

apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: myclaim

spec:
  accessModes:
    - ReadWriteOnce          # Volume can be mounted read-write on ONE node
  resources:
    requests:
      storage: 1Gi           # PV must have at least 8Gi capacity
  storageClassName: slow     # PVC will only consider PVs with SC = slow
  selector:                  # Extra filter beyond SC + size
    matchLabels:
      release: "stable"      # PV must have label release=stable
    matchExpressions:        # Advanced label matching
      - key: environment
        operator: In
        values:
          - dev              # PV must have environment=dev
```

```yaml
apiVersion: v1
kind: PersistentVolume
metadata:
  name: fci-solution-logs

  labels:                    # These labels satisfy the PVC selector
    component: solution
    function: logs

spec:
  capacity:
    storage: 1Gi             # Must be >= PVC request

  accessModes:
    - ReadWriteOnce          # Must support PVC mode

  persistentVolumeReclaimPolicy: Retain  # Data kept after PVC deletion

  storageClassName: slow     # Must match PVC SC

  nfs:
    path: /fci-solution       # Real storage path on NFS server
    server: HOSTNAME          # NFS server address

---

apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: solution-logs-claim

spec:
  accessModes:
    - ReadWriteOnce          # Single-node read-write

  storageClassName: slow     # Only PVs in SC 'slow' are considered

  resources:
    requests:
      storage: 1Gi           # Needs 1Gi or more

  selector:
    matchLabels:
      component: solution    # Only PVs labeled for solution app
      function: logs         # Only PVs meant for logs
```
