### ETCD backup
* Take backup of etcd datastore.
* Not resources yamls(already versioned controlled)
* Backup on PVs

### Why/when backup?
* Before upgrade.
* Schedule --> Policy

### Self managed K8S cluster
* Got access of etcd, directly interact with it.
* Tools --> `etcdctl`

### Managed K8S services
* etcd is not accessiable
* user tools like `valero`(third party tool) to interaact with etcd and take backup.

```bash
etcdctl -h

etcdctl --endpoints 127.0.0.1:2379 \
  --cert=/etc/kubernetes/pki/etcd/server.crt \
  --key=/etc/kubernetes/pki/etcd/server.key \
  --cacert=/etc/kubernetes/pki/etcd/ca.crt \
  member list

# 264d7b068180479b, started, controlplane, https://172.30.1.2:2380, https://172.30.1.2:2379, false

etcdctl --endpoints=http://10.0.0.2,http://10.0.0.3 member list
#Etcd return a unique ip for each member, etcd user these unique id internally.

etcdctl member remove 8211f1d0f64f3269
etcdctl member add member4 --peer-urls=http://10.0.0.4:2380

ETCDCTL_API=3 etcdctl --endpoints=https://127.0.0.1:2379 \
  --cacert=<trusted-ca-file> --cert=<cert-file> --key=<key-file> \
  snapshot save <backup-file-location>

export ETCDCTL_API=3 

etcdctl --endpoints=https://127.0.0.1:2379 \
  --cacert=/etc/kubernetes/pki/etcd/ca.crt --cert=/etc/kubernetes/pki/etcd/server.crt --key=/etc/kubernetes/pki/etcd/server.key \
  snapshot save /opt/etcd-backup.db

du -sh etcd-backup.db

etcdutl snapshot status /opt/etcd-backup.db --write-out=table
+----------+----------+------------+------------+---------+
|   HASH   | REVISION | TOTAL KEYS | TOTAL SIZE | VERSION |
+----------+----------+------------+------------+---------+
| de8aaae0 |     5526 |        673 |     4.9 MB |   3.6.0 |
+----------+----------+------------+------------+---------+

mv /etc/kubernetes/manifests/kube-* /tmp

etcdutl --data-dir /var/lib/etcd-backup-restored snapshot restore /opt/etcd-backup.db

grep -A5 -n "data-dir" /etc/kubernetes/manifests/etcd.yaml
grep -A5 -n "volumeMounts" /etc/kubernetes/manifests/etcd.yaml
grep -A5 -n "volumes:" /etc/kubernetes/manifests/etcd.yaml

sudo sed -i 's|--data-dir=.*|--data-dir=/var/lib/etcd-backup-restored|' /etc/kubernetes/manifests/etcd.yaml
sudo sed -i 's|mountPath: /var/lib/etcd|mountPath: /var/lib/etcd-backup-restored|' /etc/kubernetes/manifests/etcd.yaml
sudo sed -i 's|path: /var/lib/etcd|path: /var/lib/etcd-backup-restored|' /etc/kubernetes/manifests/etcd.

# or

sudo sed -i 's|/var/lib/etcd|/var/lib/etcd-backup-restored|g' /etc/kubernetes/manifests/etcd.yaml

mv /tmp/kube-* /etc/kubernetes/manifests/
sudo systemctl restart kubelet

```
