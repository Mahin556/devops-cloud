```yaml
apiVersion: v1
kind: Pod
metadata:
  name: silly-demo
  labels:
    app: silly-demo

spec:
  containers:
  - name: silly-demo
    image: ghcr.io/vfarcic/silly-demo:1.4.235-alpine

    ports:
      - containerPort: 8080

    readinessProbe:
      httpGet:
        path: /
        port: 8080

    env:
      - name: DB
        value: fs   # Store data on filesystem

    volumeMounts:
      - name: silly-cache
        mountPath: /cache   # App writes here

  volumes:
    - name: silly-cache
      emptyDir: {}          # Ephemeral storage
```
